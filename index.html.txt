<script>

/* ================= 변수 ================= */

let userName = "";
let heartTimer = null;


/* ================= 화면 전환 ================= */

function hideAllScreens() {

    document.getElementById("firstScreen").style.display = "none";
    document.getElementById("questionScreen").style.display = "none";
    document.getElementById("sadScreen").style.display = "none";
    document.getElementById("secondQuestionScreen").style.display = "none";
    document.getElementById("loveScreen").style.display = "none";
}


/* ================= 이름 확인 ================= */

function startQuestion() {

    const input = document.getElementById("nameInput");

    const value = input.value.trim();

    if (value.length === 0) {

        alert("이름을 입력해주세요!");

        input.focus();

        return;
    }

    userName = value;

    hideAllScreens();

    document.getElementById("questionScreen").style.display = "flex";
}


/* ================= 응 ================= */

function yesAnswer() {

    hideAllScreens();

    const loveScreen =
        document.getElementById("loveScreen");

    loveScreen.style.display = "flex";

    document.getElementById("loveName").textContent =
        "저는 " + userName + "을(를) 포함한 모두를 좋아합니다 ❤️";

    startHearts();
}


/* ================= 첫 번째 아니 ================= */

function noAnswerFirst() {

    hideAllScreens();

    document.getElementById("sadScreen").style.display = "flex";
}


/* ================= 다시 선택 ================= */

function goSecondQuestion() {

    hideAllScreens();

    const noButton =
        document.getElementById("noButton2");

    /* 위치를 원래대로 초기화 */
    noButton.style.position = "relative";
    noButton.style.left = "";
    noButton.style.top = "";
    noButton.style.transform = "";

    document.getElementById("secondQuestionScreen").style.display = "flex";
}


/* ================= 두 번째 아니 ================= */

function runAwayNoButton() {

    const button =
        document.getElementById("noButton2");

    const buttonWidth = button.offsetWidth;
    const buttonHeight = button.offsetHeight;

    const maxX =
        window.innerWidth - buttonWidth - 20;

    const maxY =
        window.innerHeight - buttonHeight - 20;

    const x =
        Math.max(10, Math.random() * maxX);

    const y =
        Math.max(10, Math.random() * maxY);

    button.style.position = "fixed";
    button.style.left = x + "px";
    button.style.top = y + "px";

    button.style.transform =
        "rotate(" +
        (Math.random() * 30 - 15) +
        "deg)";

    document.getElementById("yesButton2").style.transform =
        "scale(1.12)";
}


/* ================= 하트 ================= */

function startHearts() {

    if (heartTimer !== null) {
        clearInterval(heartTimer);
    }

    heartTimer = setInterval(function() {

        const heart =
            document.createElement("div");

        heart.className = "heart";

        const heartList = [
            "❤️",
            "💕",
            "💗",
            "💖",
            "💘",
            "💝"
        ];

        heart.textContent =
            heartList[
                Math.floor(
                    Math.random() * heartList.length
                )
            ];

        heart.style.left =
            Math.random() * 100 + "vw";

        heart.style.fontSize =
            (20 + Math.random() * 35) + "px";

        heart.style.animationDuration =
            (3 + Math.random() * 4) + "s";

        document
            .getElementById("loveScreen")
            .appendChild(heart);

        setTimeout(function() {

            heart.remove();

        }, 7000);

    }, 180);
}


/* ================= 버튼 연결 ================= */

/*
   onclick 대신 JavaScript에서 직접 연결.
   모바일 브라우저에서도 안정적으로 작동하도록 함.
*/

document
    .getElementById("okButton")
    .addEventListener("click", startQuestion);

document
    .getElementById("yesButton1")
    .addEventListener("click", yesAnswer);

document
    .getElementById("noButton1")
    .addEventListener("click", noAnswerFirst);

document
    .getElementById("retryButton")
    .addEventListener("click", goSecondQuestion);

document
    .getElementById("yesButton2")
    .addEventListener("click", yesAnswer);

document
    .getElementById("noButton2")
    .addEventListener("click", runAwayNoButton);


/* ================= 엔터키 ================= */

document
    .getElementById("nameInput")
    .addEventListener("keydown", function(event) {

        if (event.key === "Enter") {

            event.preventDefault();

            startQuestion();
        }
    });
